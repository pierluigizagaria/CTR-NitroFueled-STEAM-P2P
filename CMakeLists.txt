cmake_minimum_required(VERSION 3.20)
project(ctrnf-p2p VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_path(OSW_INCLUDE_DIR orbisteamworks/osw.h REQUIRED)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_library(bridge_core src/config.cpp src/http.cpp src/platform.cpp src/provider.cpp src/host.cpp)
target_include_directories(bridge_core PUBLIC src ${OSW_INCLUDE_DIR} ${JSON_INCLUDE_DIR})
target_link_libraries(bridge_core PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
  target_link_libraries(bridge_core PUBLIC ws2_32 iphlpapi)
  target_compile_definitions(bridge_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
if(MSVC)
  target_compile_options(bridge_core PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(bridge_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(ctrnf-p2p src/main.cpp)
target_link_libraries(ctrnf-p2p PRIVATE bridge_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(bridge_tests tests/contracts.cpp)
  target_link_libraries(bridge_tests PRIVATE bridge_core)
  add_test(NAME bridge-contracts COMMAND bridge_tests)
  add_subdirectory(tests)
endif()
