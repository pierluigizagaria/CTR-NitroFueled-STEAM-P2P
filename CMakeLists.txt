cmake_minimum_required(VERSION 3.20)
project(CTR-NitroFueled-STEAM-P2P VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_path(OSW_INCLUDE_DIR orbisteamworks/osw.h REQUIRED)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_subdirectory(third_party/expat)
add_library(bridge_core src/config.cpp src/http.cpp src/platform.cpp src/provider.cpp src/host.cpp src/psn.cpp)
target_include_directories(bridge_core PUBLIC src ${OSW_INCLUDE_DIR} ${JSON_INCLUDE_DIR})
target_link_libraries(bridge_core PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
target_link_libraries(bridge_core PRIVATE bridge_expat)
if(WIN32)
  target_link_libraries(bridge_core PUBLIC ws2_32 iphlpapi bcrypt)
  target_compile_definitions(bridge_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
if(MSVC)
  target_compile_options(bridge_core PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(bridge_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(bridge_app STATIC src/main.cpp)
target_link_libraries(bridge_app PUBLIC bridge_core)
if(WIN32)
  add_executable(ctrnf-p2p WIN32 src/desktop_windows.cpp)
  target_link_libraries(ctrnf-p2p PRIVATE bridge_app shell32 user32)
  target_compile_definitions(ctrnf-p2p PRIVATE UNICODE _UNICODE)
  if(MINGW)
    target_link_options(ctrnf-p2p PRIVATE -municode)
  endif()
  add_executable(ctrnf-p2p-cli src/cli_main.cpp)
  target_link_libraries(ctrnf-p2p-cli PRIVATE bridge_app)
  set(BRIDGE_CLI_TARGET ctrnf-p2p-cli)
else()
  add_executable(ctrnf-p2p src/cli_main.cpp)
  target_link_libraries(ctrnf-p2p PRIVATE bridge_app)
  set(BRIDGE_CLI_TARGET ctrnf-p2p)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(bridge_tests tests/contracts.cpp)
  target_link_libraries(bridge_tests PRIVATE bridge_core)
  add_test(NAME bridge-contracts COMMAND bridge_tests)
  add_subdirectory(tests)
endif()
